def Approx252.IsSofic.{u_1} : (G : Type u_1) → [Group G] → Prop :=
fun G [Group G] =>
  ∀ (F : Finset G) (ε : ℝ),
    0 < ε →
      ∃ n φ,
        (∀ g ∈ F, ∀ h ∈ F, Approx252.hammingDist (φ (g * h)) (φ g * φ h) ≤ ε) ∧
          ∀ g ∈ F, g ≠ 1 → 1 - ε ≤ Approx252.hammingDist (φ g) 1
def Approx252.hammingDist : {n : ℕ} → Equiv.Perm (Fin n) → Equiv.Perm (Fin n) → ℝ :=
fun {n} σ τ => ↑(Approx252.dis σ τ) / ↑n
def Approx252.dis.{u_1} : {Ω : Type u_1} → [Fintype Ω] → [DecidableEq Ω] → Equiv.Perm Ω → Equiv.Perm Ω → ℕ :=
fun {Ω} [Fintype Ω] [DecidableEq Ω] σ τ => (σ⁻¹ * τ).support.card
def Approx252.RectanglesAlmostFree.{u_1, u_2} : {Ω : Type u_1} →

fun {Ω} [Fintype Ω] [DecidableEq Ω] {ι} x e =>
  ∀ (u u' v v' : ι), u ≠ u' → v ≠ v' → Approx252.fixCount (x u v * (x u' v)⁻¹ * x u' v' * (x u v')⁻¹) ≤ e
